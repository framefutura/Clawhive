```typescript
import { spawn, ChildProcess } from 'node:child_process'
import { EventEmitter } from 'node:events'
import WebSocket from 'ws'
import { app } from 'electron'

const GATEWAY_DEFAULT_PORT = 18792
const GATEWAY_START_TIMEOUT = 15000

export interface GatewayConfig {
  port?: number
  authToken?: string
}

export interface GatewayEvent {
  type: 'connected' | 'disconnected' | 'message' | 'error'
  data?: unknown
}

export class GatewayManager extends EventEmitter {
  private process: ChildProcess | null = null
  private ws: WebSocket | null = null
  private port: number
  private authToken: string
  private reconnectTimer: NodeJS.Timeout | null = null

  constructor(config: GatewayConfig = {}) {
    super()
    this.port = config.port ?? GATEWAY_DEFAULT_PORT
    this.authToken = config.authToken ?? ''
  }

  async start(): Promise<void> {
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        reject(new Error(`Gateway failed to start within ${GATEWAY_START_TIMEOUT}ms`))
      }, GATEWAY_START_TIMEOUT)

      // Spawn gateway as child process
      this.process = spawn('npx', ['openclaw', 'gateway', '--port', String(this.port)], {
        stdio: ['pipe', 'pipe', 'pipe'],
        env: {
          ...process.env,
          OPENCLAW_TOKEN: this.authToken,
        },
      })

      this.process.on('error', (err) => {
        clearTimeout(timeout)
        reject(err)
      })

      this.process.stderr?.on('data', (data: Buffer) => {
        const msg = data.toString()
        if (msg.includes('listening') || msg.includes('Gateway running')) {
          clearTimeout(timeout)
          this.connectWebSocket().then(resolve).catch(reject)
        }
      })
    })
  }

  private async connectWebSocket(): Promise<void> {
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        reject(new Error('WebSocket connection timeout'))
      }, 5000)

      this.ws = new WebSocket(`ws://localhost:${this.port}`)

      this.ws.on('open', () => {
        clearTimeout(timeout)
        // Send connect request
        this.ws?.send(JSON.stringify({
          type: 'req',
          action: 'connect',
          authToken: this.authToken,
        }))
        this.emit('event', { type: 'connected' } satisfies GatewayEvent)
        resolve()
      })

      this.ws.on('message', (data: WebSocket.Data) => {
        try {
          const msg = JSON.parse(data.toString())
          this.emit('event', { type: 'message', data: msg } satisfies GatewayEvent)
        } catch {
          // ignore parse errors
        }
      })

      this.ws.on('close', () => {
        this.emit('event', { type: 'disconnected' } satisfies GatewayEvent)
      })

      this.ws.on('error', (err) => {
        clearTimeout(timeout)
        reject(err)
      })
    })
  }

  send(action: string, payload: unknown): void {
    if (this.ws?.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify({ type: 'req', action, ...payload }))
    }
  }

  stop(): void {
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer)
    this.ws?.close()
    this.process?.kill()
    this.process = null
    this.ws = null
  }
}
```

```typescript
export interface Session {
  id: string
  agentId: string
  modelConfig: ModelConfig
  createdAt: number
  messages: ChatMessage[]
}

export interface ModelConfig {
  provider: 'anthropic' | 'openai' | 'ollama'
  model: string
  apiKey?: string
  baseUrl?: string
}

export interface ChatMessage {
  id: string
  role: 'user' | 'assistant'
  content: string
  attachments?: Attachment[]
  timestamp: number
}

export interface Attachment {
  name: string
  type: string
  path: string
}

export interface SessionStore {
  create(agentId: string, modelConfig: ModelConfig): Session
  list(): Session[]
  get(id: string): Session | null
  delete(id: string): void
  addMessage(sessionId: string, message: Omit<ChatMessage, 'id' | 'timestamp'>): void
}

class InMemorySessionStore implements SessionStore {
  private sessions = new Map<string, Session>()

  create(agentId: string, modelConfig: ModelConfig): Session {
    const session: Session = {
      id: crypto.randomUUID(),
      agentId,
      modelConfig,
      createdAt: Date.now(),
      messages: [],
    }
    this.sessions.set(session.id, session)
    return session
  }

  list(): Session[] {
    return Array.from(this.sessions.values())
  }

  get(id: string): Session | null {
    return this.sessions.get(id) ?? null
  }

  delete(id: string): void {
    this.sessions.delete(id)
  }

  addMessage(sessionId: string, message: Omit<ChatMessage, 'id' | 'timestamp'>): void {
    const session = this.sessions.get(sessionId)
    if (session) {
      session.messages.push({
        ...message,
        id: crypto.randomUUID(),
        timestamp: Date.now(),
      })
    }
  }
}

export const sessionStore = new InMemorySessionStore()
```

```typescript
import { useState, useEffect, useCallback } from 'react'

export function useGateway() {
  const [connected, setConnected] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const connect = useCallback(async (gatewayUrl: string) => {
    try {
      setError(null)
      await window.clawhive.connect(gatewayUrl)
      setConnected(true)
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Connection failed')
      setConnected(false)
    }
  }, [])

  const disconnect = useCallback(() => {
    window.clawhive.disconnect()
    setConnected(false)
  }, [])

  useEffect(() => {
    const cleanup = window.clawhive.onGatewayEvent((event) => {
      if ((event as { type: string }).type === 'connected') setConnected(true)
      if ((event as { type: string }).type === 'disconnected') setConnected(false)
    })
    return cleanup
  }, [])

  return { connected, error, connect, disconnect }
}
```

```typescript
import { useState, useEffect, useCallback } from 'react'
import type { Session } from '../../main/session'

export function useSession() {
  const [sessions, setSessions] = useState<Session[]>([])
  const [activeSession, setActiveSession] = useState<Session | null>(null)

  const createSession = useCallback(async (agentId: string, modelConfig: unknown) => {
    const session = await window.clawhive.createSession(agentId, modelConfig as Parameters<typeof window.clawhive.createSession>[1])
    setSessions(prev => [...prev, session])
    setActiveSession(session)
    return session
  }, [])

  const deleteSession = useCallback(async (sessionId: string) => {
    await window.clawhive.deleteSession(sessionId)
    setSessions(prev => prev.filter(s => s.id !== sessionId))
    if (activeSession?.id === sessionId) setActiveSession(null)
  }, [activeSession])

  const loadSessions = useCallback(async () => {
    const list = await window.clawhive.getSessions()
    setSessions(list)
  }, [])

  useEffect(() => {
    loadSessions()
  }, [loadSessions])

  return { sessions, activeSession, createSession, deleteSession, setActiveSession, loadSessions }
}
```

```typescript
import { useState, useCallback } from 'react'

export interface ChatMessage {
  id: string
  role: 'user' | 'assistant'
  content: string
  timestamp: number
  attachments?: File[]
}

export function useChatStore() {
  const [messages, setMessages] = useState<ChatMessage[]>([])
  const [isWorking, setIsWorking] = useState(false)

  const sendMessage = useCallback(async (content: string) => {
    if (!content.trim()) return

    // Add user message immediately
    const userMsg: ChatMessage = {
      id: crypto.randomUUID(),
      role: 'user',
      content,
      timestamp: Date.now(),
    }
    setMessages(prev => [...prev, userMsg])
    setIsWorking(true)

    try {
      // Listen for streaming response
      const cleanup = window.clawhive.onMessage((msg) => {
        const typedMsg = msg as { id: string; role: string; content: string }
        setMessages(prev => {
          const last = prev[prev.length - 1]
          if (last?.role === 'assistant' && last.id === typedMsg.id) {
            return [...prev.slice(0, -1), { ...last, content: typedMsg.content }]
          }
          return [...prev, { id: typedMsg.id, role: typedMsg.role as 'user' | 'assistant', content: typedMsg.content, timestamp: Date.now() }]
        })
      })

      await window.clawhive.sendMessage('default', content)
      cleanup()
    } finally {
      setIsWorking(false)
    }
  }, [])

  return { messages, isWorking, sendMessage }
}
```

```typescript
import React, { useState, useRef, useCallback } from 'react'

interface ChatInputProps {
  onSend: (content: string) => void
  disabled?: boolean
}

export function ChatInput({ onSend, disabled }: ChatInputProps) {
  const [value, setValue] = useState('')
  const textareaRef = useRef<HTMLTextAreaElement>(null)

  const handleSend = useCallback(() => {
    if (value.trim() && !disabled) {
      onSend(value.trim())
      setValue('')
      textareaRef.current?.focus()
    }
  }, [value, disabled, onSend])

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
      e.preventDefault()
      handleSend()
    }
  }

  return (
    <div className="chat-input-area">
      <div className="input-wrapper">
        <textarea
          ref={textareaRef}
          className="chat-input"
          value={value}
          onChange={(e) => setValue(e.target.value)}
          onKeyDown={handleKeyDown}
          placeholder="Type a message..."
          disabled={disabled}
          rows={1}
          aria-label="Chat message input"
        />
        <button
          className="send-button"
          onClick={handleSend}
          disabled={disabled || !value.trim()}
          aria-label="Send message"
        >
          Send Message
        </button>
      </div>
      <div className="helper-text">⌘↵ to send · Supports drag-drop, paste, or click to upload</div>
    </div>
  )
}
```

```typescript
import React, { useState } from 'react'

export type Provider = 'anthropic' | 'openai' | 'ollama'

interface ModelInfo {
  id: string
  name: string
  provider: Provider
}

const PROVIDER_LABELS: Record<Provider, string> = {
  anthropic: 'Anthropic',
  openai: 'OpenAI',
  ollama: 'Ollama',
}

const DEFAULT_MODELS: Record<Provider, string[]> = {
  anthropic: ['claude-sonnet-4-20250514', 'claude-opus-4-20250514', 'claude-3-5-sonnet-latest'],
  openai: ['gpt-4o', 'gpt-4o-mini', 'gpt-4-turbo'],
  ollama: [], // Fetched dynamically
}

interface ModelPickerProps {
  selectedProvider: Provider
  selectedModel: string
  onProviderChange: (provider: Provider) => void
  onModelChange: (model: string) => void
  onApiKeyChange: (provider: Provider, key: string) => void
}

export function ModelPicker({
  selectedProvider,
  selectedModel,
  onProviderChange,
  onModelChange,
  onApiKeyChange,
}: ModelPickerProps) {
  const [open, setOpen] = useState(false)
  const [apiKeys, setApiKeys] = useState<Partial<Record<Provider, string>>>({})

  const handleApiKeySave = (provider: Provider) => {
    const key = apiKeys[provider]
    if (key) onApiKeyChange(provider, key)
    setOpen(false)
  }

  return (
    <div style={{ position: 'relative' }}>
      <button
        onClick={() => setOpen(!open)}
        style={{
          display: 'flex',
          alignItems: 'center',
          gap: '6px',
          padding: '6px 12px',
          background: 'var(--bg-tertiary)',
          border: '1px solid var(--border)',
          borderRadius: '6px',
          color: 'var(--text-primary)',
          fontSize: '13px',
          cursor: 'pointer',
        }}
        aria-label="Select model"
      >
        <span style={{ fontWeight: 500 }}>{PROVIDER_LABELS[selectedProvider]}</span>
        <span>·</span>
        <span>{selectedModel}</span>
        <span>▾</span>
      </button>

      {open && (
        <div
          style={{
            position: 'absolute',
            top: '100%',
            left: 0,
            marginTop: '4px',
            background: 'var(--bg-secondary)',
            border: '1px solid var(--border)',
            borderRadius: '8px',
            minWidth: '280px',
            zIndex: 50,
            boxShadow: '0 8px 24px rgba(0,0,0,0.3)',
          }}
        >
          {(['anthropic', 'openai', 'ollama'] as Provider[]).map((provider) => (
            <div key={provider} style={{ padding: '8px 12px', borderBottom: '1px solid var(--border)' }}>
              <div style={{ fontSize: '12px', fontWeight: 600, color: 'var(--text-secondary)', marginBottom: '6px' }}>
                {PROVIDER_LABELS[provider]}
              </div>
              {provider === selectedProvider ? (
                <select
                  value={selectedModel}
                  onChange={(e) => onModelChange(e.target.value)}
                  style={{
                    width: '100%',
                    padding: '6px 8px',
                    background: 'var(--bg-tertiary)',
                    border: '1px solid var(--border)',
                    borderRadius: '4px',
                    color: 'var(--text-primary)',
                    fontSize: '13px',
                  }}
                >
                  <option value="">Select model</option>
                  {DEFAULT_MODELS[provider].map((m) => (
                    <option key={m} value={m}>{m}</option>
                  ))}
                </select>
              ) : (
                <button
                  onClick={() => onProviderChange(provider)}
                  style={{
                    width: '100%',
                    padding: '6px 8px',
                    background: 'transparent',
                    border: '1px solid var(--border)',
                    borderRadius: '4px',
                    color: 'var(--text-secondary)',
                    fontSize: '13px',
                    cursor: 'pointer',
                  }}
                >
                  Use {PROVIDER_LABELS[provider]}
                </button>
              )}
              <div style={{ marginTop: '6px' }}>
                <input
                  type="password"
                  placeholder={`${PROVIDER_LABELS[provider]} API key`}
                  value={apiKeys[provider] || ''}
                  onChange={(e) => setApiKeys(prev => ({ ...prev, [provider]: e.target.value }))}
                  style={{
                    width: '100%',
                    padding: '6px 8px',
                    background: 'var(--bg-tertiary)',
                    border: '1px solid var(--border)',
                    borderRadius: '4px',
                    color: 'var(--text-primary)',
                    fontSize: '13px',
                    marginBottom: '4px',
                  }}
                />
                <button
                  onClick={() => handleApiKeySave(provider)}
                  style={{
                    width: '100%',
                    padding: '4px 8px',
                    background: 'var(--accent)',
                    border: 'none',
                    borderRadius: '4px',
                    color: 'white',
                    fontSize: '12px',
                    cursor: 'pointer',
                  }}
                >
                  Save API Key
                </button>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  )
}
```
