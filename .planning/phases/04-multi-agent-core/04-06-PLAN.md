export function createTeam(team: Omit<DbTeam, 'created_at'>): void {
  if (!db) throw new Error('Database not initialized')
  const now = Date.now()
  db.run(
    'INSERT INTO teams (id, name, leader_id, department, workspace_id, created_at) VALUES (?, ?, ?, ?, ?, ?)',
    [team.id, team.name, team.leader_id, team.department ?? null, team.workspace_id ?? null, now]
  )
  saveDatabase().catch(console.error)
}

export function getTeam(id: string): DbTeam | null {
  if (!db) throw new Error('Database not initialized')
  const stmt = db.prepare('SELECT * FROM teams WHERE id = ?')
  stmt.bind([id])
  let result: DbTeam | null = null
  if (stmt.step()) {
    result = stmt.getAsObject() as unknown as DbTeam
  }
  stmt.free()
  return result
}

export function listTeams(): DbTeam[] {
  if (!db) throw new Error('Database not initialized')
  const stmt = db.prepare('SELECT * FROM teams ORDER BY created_at DESC')
  const results: DbTeam[] = []
  while (stmt.step()) {
    results.push(stmt.getAsObject() as unknown as DbTeam)
  }
  stmt.free()
  return results
}

export function deleteTeam(id: string): void {
  if (!db) throw new Error('Database not initialized')
  db.run('DELETE FROM teams WHERE id = ?', [id])
  saveDatabase().catch(console.error)
}

export function addTeamMember(teamId: string, agentId: string): void {
  if (!db) throw new Error('Database not initialized')
  db.run(
    'INSERT OR IGNORE INTO team_members (team_id, agent_id, joined_at) VALUES (?, ?, ?)',
    [teamId, agentId, Date.now()]
  )
  saveDatabase().catch(console.error)
}

export function removeTeamMember(teamId: string, agentId: string): void {
  if (!db) throw new Error('Database not initialized')
  db.run('DELETE FROM team_members WHERE team_id = ? AND agent_id = ?', [teamId, agentId])
  saveDatabase().catch(console.error)
}

export function listTeamMembers(teamId: string): string[] {
  if (!db) throw new Error('Database not initialized')

  }

```typescript
import fs from 'node:fs/promises'
import path from 'node:path'
import { getDataPath } from './storage.js'

/**
 * Directory structure:
 *
 * ~/.clawhive/
 *   agents/
 *     <agent-id>/           # Agent isolated storage (private to agent)
 *   teams/
 *     <team-id>/             # Team shared storage
 *       shared_files/        # Shared files accessible to all team members
 *       memory/               # Shared team memory/context
 *       output/               # Team task outputs and artifacts
 *       members/<agent-id>/  # Agent's own subfolder within team (private-ish, scoped to team)
 */

export const CLAWHIVE_DIR = '.clawhive'
export const AGENTS_DIR = 'agents'
export const TEAMS_DIR = 'teams'
export const TEAM_SUBDIRS = ['shared_files', 'memory', 'output'] as const
export const AGENT_IN_TEAM_SUBDIR = 'members' // subdir name for agent's scoped folder inside team dir

function getBaseDir(): string {
  const dataPath = getDataPath()
  if (!dataPath) {
    throw new Error('Data path not configured — call loadDatabase() first')
  }
  return dataPath
}

/** Path: ~/.clawhive/agents/<agentId>/ */
export function getAgentStoragePath(agentId: string): string {
  return path.join(getBaseDir(), CLAWHIVE_DIR, AGENTS_DIR, agentId)
}

/** Path: ~/.clawhive/teams/<teamId>/ */
export function getTeamStoragePath(teamId: string): string {
  return path.join(getBaseDir(), CLAWHIVE_DIR, TEAMS_DIR, teamId)
}

/** Path: ~/.clawhive/teams/<teamId>/members/<agentId>/ */
export function getAgentTeamSubdirPath(teamId: string, agentId: string): string {
  return path.join(getTeamStoragePath(teamId), AGENT_IN_TEAM_SUBDIR, agentId)
}

/**
 * Create an agent's isolated storage directory.
 * Called automatically when an agent is created.
 */
export async function createAgentStorageDir(agentId: string): Promise<string> {
  const dir = getAgentStoragePath(agentId)
  await fs.mkdir(dir, { recursive: true })
  return dir
}

/**
 * Create a team's shared storage directory with all required subdirectories.
 * Called automatically when a team is created.
 *
 * Structure:
 *   ~/.clawhive/teams/<teamId>/
 *     shared_files/
 *     memory/
 *     output/
 *     members/<agentId>/   (created on demand via ensureAgentTeamSubdir)
 */
export async function createTeamStorageDir(teamId: string): Promise<string> {
  const teamDir = getTeamStoragePath(teamId)
  await fs.mkdir(teamDir, { recursive: true })
  for (const subdir of TEAM_SUBDIRS) {
    await fs.mkdir(path.join(teamDir, subdir), { recursive: true })
  }
  return teamDir
}

/**
 * Ensure an agent has its scoped subfolder inside a team directory.
 * Called when an agent joins a team.
 */
export async function ensureAgentTeamSubdir(teamId: string, agentId: string): Promise<string> {
  const dir = getAgentTeamSubdirPath(teamId, agentId)
  await fs.mkdir(dir, { recursive: true })
  return dir
}

/**
 * Delete an agent's isolated storage directory.
 * Should be called when an agent is deleted.
 */
export async function deleteAgentStorageDir(agentId: string): Promise<void> {
  const dir = getAgentStoragePath(agentId)
  try {
    await fs.rm(dir, { recursive: true, force: true })
  } catch {
    // Ignore if already deleted
  }
}

/**
 * Delete a team's storage directory and all its contents.
 * Should be called when a team is deleted.
 */
export async function deleteTeamStorageDir(teamId: string): Promise<void> {
  const dir = getTeamStoragePath(teamId)
  try {
    await fs.rm(dir, { recursive: true, force: true })
  } catch {
    // Ignore if already deleted
  }
}

/**
 * Remove an agent's scoped subfolder from a team directory.
 * Should be called when an agent leaves a team.
 */
export async function removeAgentFromTeamDir(teamId: string, agentId: string): Promise<void> {
  const dir = getAgentTeamSubdirPath(teamId, agentId)
  try {
    await fs.rm(dir, { recursive: true, force: true })
  } catch {
    // Ignore if already deleted
  }
}

/**
 * Check if an agent's storage directory exists.
 */
export async function agentStorageExists(agentId: string): Promise<boolean> {
  try {
    const dir = getAgentStoragePath(agentId)
    await fs.access(dir)
    return true
  } catch {
    return false
  }
}

/**
 * Check if a team's storage directory exists.
 */
export async function teamStorageExists(teamId: string): Promise<boolean> {
  try {
    const dir = getTeamStoragePath(teamId)
    await fs.access(dir)
    return true
  } catch {
    return false
  }
}
```
