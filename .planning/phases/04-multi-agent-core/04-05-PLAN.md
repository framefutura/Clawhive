import fs from 'node:fs/promises'
import path from 'node:path'

// Agent doc file names
export const AGENT_DOCS = [
  'soul.md',        // Agent's identity, values, decision-making principles
  'heartbeat.md',   // Task execution loop, interval config, behavior
  'tools.md',       // Allowed tools, usage patterns, constraints
  'agents.md',      // Relationships with other agents, delegation rules
  'interaction.md', // How this agent communicates, tone, escalation paths
] as const

export type AgentDocName = typeof AGENT_DOCS[number]

export class AgentDocsManager {
  constructor(private dataPath: string) {}

  private getAgentDir(agentId: string): string {
    return path.join(this.dataPath, 'agents', agentId)
  }

  // List all doc filenames and their existence + modified time for an agent
  async listDocs(agentId: string): Promise<Array<{ name: string; exists: boolean; size: number; modifiedAt: number }>> {
    const dir = this.getAgentDir(agentId)
    const results = []
    for (const doc of AGENT_DOCS) {
      const filePath = path.join(dir, doc)
      try {
        const stat = await fs.stat(filePath)
        results.push({ name: doc, exists: true, size: stat.size, modifiedAt: stat.mtimeMs })
      } catch {
        results.push({ name: doc, exists: false, size: 0, modifiedAt: 0 })
      }
    }
    return results
  }

  // Read a specific doc
  async readDoc(agentId: string, docName: string): Promise<string | null> {
    if (!AGENT_DOCS.includes(docName as AgentDocName)) return null
    const filePath = path.join(this.getAgentDir(agentId), docName)
    try {
      return await fs.readFile(filePath, 'utf-8')
    } catch {
      return null // File does not exist yet
    }
  }

  // Write a doc (creates dir if needed)
  async writeDoc(agentId: string, docName: string, content: string): Promise<void> {
    if (!AGENT_DOCS.includes(docName as AgentDocName)) throw new Error(`Invalid doc: ${docName}`)
    const dir = this.getAgentDir(agentId)
    await fs.mkdir(dir, { recursive: true })
    await fs.writeFile(path.join(dir, docName), content, 'utf-8')
  }

  // Delete a doc
  async deleteDoc(agentId: string, docName: string): Promise<void> {
    if (!AGENT_DOCS.includes(docName as AgentDocName)) return
    const filePath = path.join(this.getAgentDir(agentId), docName)
    await fs.unlink(filePath).catch(() => {}) // Ignore if not found
  }

  // Initialize agent docs dir with default doc content
  async initializeAgentDocs(agentId: string, defaults: Partial<Record<AgentDocName, string>>): Promise<void> {
    const dir = this.getAgentDir(agentId)
    await fs.mkdir(dir, { recursive: true })
    const defaultContent: Record<AgentDocName, string> = {
      'soul.md': defaults['soul.md'] ?? `# Soul\n\nIdentity and values for this agent.\n`,
      'heartbeat.md': defaults['heartbeat.md'] ?? `# Heartbeat\n\nTask execution loop configuration.\n`,
      'tools.md': defaults['tools.md'] ?? `# Tools\n\nAllowed tools for this agent.\n`,
      'agents.md': defaults['agents.md'] ?? `# Agents\n\nRelationships with other agents.\n`,
      'interaction.md': defaults['interaction.md'] ?? `# Interaction\n\nCommunication style and escalation paths.\n`,
    }
    for (const [doc, content] of Object.entries(defaultContent)) {
      const filePath = path.join(dir, doc)
      try {
        await fs.access(filePath)
        // File already exists, do not overwrite
      } catch {
        await fs.writeFile(filePath, content as string, 'utf-8')
      }
    }
  }
}

import React, { useState } from 'react'

export type EditorMode = 'split-pane' | 'tabbed' | 'card'

export interface InteractionEditorProps {
  files: Array<{ name: string; content: string; exists: boolean }>
  activeFile: string
  onChangeFile: (name: string) => void
  onSave: (name: string, content: string) => void
  onModeChange?: (mode: EditorMode) => void
  mode?: EditorMode
}

// File list panel (left side of split pane, or card tabs)
function FileList({ files, activeFile, onSelect }: {
  files: Array<{ name: string; exists: boolean }>
  activeFile: string
  onSelect: (name: string) => void
}) {
  const labels: Record<string, string> = {
    'soul.md': 'Soul',
    'heartbeat.md': 'Heartbeat',
    'tools.md': 'Tools',
    'agents.md': 'Agents',
    'interaction.md': 'Interaction',
  }
  return (
    <div className="flex flex-col gap-1 p-2 min-w-[160px] bg-[var(--bg-secondary)]">
      {files.map(f => (
        <button
          key={f.name}
          onClick={() => onSelect(f.name)}
          className={`text-left px-3 py-2 rounded text-sm flex items-center gap-2 transition-colors ${
            activeFile === f.name
              ? 'bg-primary/15 text-primary border-l-2 border-primary'
              : 'hover:bg-[var(--bg-hover)] text-[var(--text-secondary)]'
          }`}
        >
          <span className={`w-2 h-2 rounded-full ${f.exists ? 'bg-green-500' : 'bg-gray-500/50'}`} />
          {labels[f.name] ?? f.name}
        </button>
      ))}
    </div>
  )
}

// Editor pane (right side of split pane)
function EditorPane({ content, onChange, onSave }: {
  content: string
  onChange: (v: string) => void
  onSave: () => void
}) {
  return (
    <div className="flex flex-col flex-1 min-h-0">
      <textarea
        value={content}
        onChange={e => onChange(e.target.value)}
        className="flex-1 p-4 font-mono text-sm bg-[var(--bg-primary)] text-[var(--text-primary)] resize-none border-none outline-none"
        spellCheck={false}
        placeholder="// Start typing..."
      />
      <div className="flex justify-end p-2 border-t border-[var(--border)]">
        <button
          onClick={onSave}
          className="px-4 py-1.5 bg-primary text-white rounded text-sm hover:opacity-90 transition-opacity"
        >
          Save
        </button>
      </div>
    </div>
  )
}

// Card-based display (expandable inline)
function CardView({ files, activeFile, onSelect, onExpand }: {
  files: Array<{ name: string; content: string; exists: boolean }>
  activeFile: string
  onSelect: (name: string) => void
  onExpand: (name: string) => void
}) {
  const labels: Record<string, string> = {
    'soul.md': 'Soul',
    'heartbeat.md': 'Heartbeat',
    'tools.md': 'Tools',
    'agents.md': 'Agents',
    'interaction.md': 'Interaction',
  }

    <div className="flex flex-col gap-2 p-3">

            activeFile === f.name
              ? 'border-primary bg-primary/5'
              : 'border-[var(--border)] hover:border-[var(--border-hover)]'

              className="text-xs text-primary hover:underline"

              Edit
            </button>
          </div>
          <p className="text-xs text-[var(--text-secondary)] mt-1 line-clamp-2 font-mono">

          </p>
        </div>
      ))}
    </div>
  )
}

export function InteractionEditor({ files, activeFile, onChangeFile, onSave, onModeChange, mode = 'split-pane' }: InteractionEditorProps) {
  const [draftContent, setDraftContent] = useState('')
  const [isDirty, setIsDirty] = useState(false)
  const [expandedCard, setExpandedCard] = useState<string | null>(null)

  const activeFileData = files.find(f => f.name === activeFile)
  const displayContent = isDirty ? draftContent : (activeFileData?.content ?? '')

  const handleSelect = (name: string) => {
    if (isDirty) {
      onSave(activeFile, draftContent)
      setIsDirty(false)
    }
    onChangeFile(name)
    setDraftContent('')
  }

  const handleChange = (v: string) => {
    setDraftContent(v)
    setIsDirty(true)
  }

  const handleSave = () => {
    onSave(activeFile, draftContent)
    setIsDirty(false)
  }

  const modeButtons: EditorMode[] = ['split-pane', 'tabbed', 'card']

  return (
    <div className="flex flex-col h-full">
      {/* Mode selector toolbar */}
      <div className="flex items-center gap-1 px-3 py-1.5 border-b border-[var(--border)] bg-[var(--bg-secondary)]">
        <span className="text-xs text-[var(--text-secondary)] mr-2">Mode:</span>
        {modeButtons.map(m => (
          <button
            key={m}
            onClick={() => onModeChange?.(m)}
            className={`px-2 py-0.5 rounded text-xs transition-colors ${
              mode === m ? 'bg-primary/20 text-primary' : 'hover:bg-[var(--bg-hover)] text-[var(--text-secondary)]'
            }`}
          >
            {m === 'split-pane' ? 'Split' : m === 'tabbed' ? 'Tabs' : 'Cards'}
          </button>
        ))}
        {isDirty && (
          <span className="ml-auto text-xs text-yellow-500">Unsaved changes</span>
        )}
      </div>

      {/* Content based on mode */}
      {mode === 'split-pane' && (
        <div className="flex flex-1 min-h-0">
          <FileList files={files} activeFile={activeFile} onSelect={handleSelect} />
          <div className="flex-1 flex flex-col min-w-0">
            <div className="px-3 py-1 text-xs text-[var(--text-secondary)] border-b border-[var(--border)]">
              {activeFile}
            </div>
            <EditorPane content={displayContent} onChange={handleChange} onSave={handleSave} />
          </div>
        </div>
      )}

      {mode === 'tabbed' && (
        <div className="flex flex-col h-full">
          {/* Tab bar */}
          <div className="flex border-b border-[var(--border)] bg-[var(--bg-secondary)] overflow-x-auto">
            {files.map(f => (
              <button
                key={f.name}
                onClick={() => handleSelect(f.name)}
                className={`px-4 py-1.5 text-xs whitespace-nowrap border-b-2 transition-colors ${
                  activeFile === f.name
                    ? 'border-primary text-primary'
                    : 'border-transparent hover:text-[var(--text-primary)]'
                }`}
              >
                {f.name}
              </button>
            ))}
          </div>
          <EditorPane content={displayContent} onChange={handleChange} onSave={handleSave} />
        </div>
      )}

      {mode === 'card' && (
        expandedCard ? (
          <div className="flex flex-col h-full">
            <div className="flex items-center gap-2 px-3 py-1.5 border-b border-[var(--border)] bg-[var(--bg-secondary)]">
              <button onClick={() => setExpandedCard(null)} className="text-xs text-primary hover:underline">
                Back to cards
              </button>
              <span className="text-xs text-[var(--text-secondary)]">/ {expandedCard}</span>
            </div>
            <EditorPane
              content={displayContent}
              onChange={handleChange}
              onSave={() => { handleSave(); setExpandedCard(null) }}
            />
          </div>
        ) : (
          <CardView
            files={files}
            activeFile={activeFile}
            onSelect={handleSelect}
            onExpand={n => { handleSelect(n); setExpandedCard(n) }}
          />
        )
      )}
    </div>
  )
}

import { InteractionEditor, type EditorMode } from './InteractionEditor'

interface DocEntry {

  exists: boolean

interface AgentDetailPanelProps {
  agentId: string
  agentName: string
  agentRole: string

export function AgentDetailPanel({ agentId, agentName, agentRole, onClose }: AgentDetailPanelProps) {
  const [docs, setDocs] = useState<DocEntry[]>([])
  const [activeFile, setActiveFile] = useState('soul.md')
  const [editorMode, setEditorMode] = useState<EditorMode>('split-pane')
  const [loading, setLoading] = useState(true)
  const [saving, setSaving] = useState(false)

    loadDocs()
  }, [agentId])

    setLoading(true)
    try {
      const docList: Array<{ name: string; exists: boolean }> = await window.clawhive.agentDocsList(agentId)
      const entries: DocEntry[] = []
      for (const doc of docList) {

        entries.push({ name: doc.name, content: content ?? '', exists: doc.exists })
      }
      setDocs(entries)
      if (entries.length > 0) setActiveFile(entries[0].name)
